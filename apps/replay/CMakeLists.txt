cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(mas_vision_replay LANGUAGES CXX)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "This tool targets Linux")
    endif()
    get_filename_component(VISION_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
        message(FATAL_ERROR "Use GCC >= 13 and its C++23 standard library")
    endif()
    if(NOT CMAKE_BUILD_TYPE)
        set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
    endif()
    if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
        set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
    endif()
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    include("${VISION_SOURCE_DIR}/cmake/Dependencies.cmake")
    add_library(buff_options INTERFACE)
    target_include_directories(buff_options INTERFACE "${VISION_SOURCE_DIR}")
    add_subdirectory("${VISION_SOURCE_DIR}/third_party" third_party)
    add_subdirectory("${VISION_SOURCE_DIR}/rm_utils" rm_utils)
    add_subdirectory("${VISION_SOURCE_DIR}/auto_buff" auto_buff)
    add_subdirectory("${VISION_SOURCE_DIR}/auto_aim" auto_aim)
endif()

add_library(vision_replay STATIC replay.cpp armor_process.cpp buff_process.cpp)
target_link_libraries(vision_replay PUBLIC auto_buff auto_aim buff_io)
add_executable(mas_vision_replay main.cpp)
target_link_libraries(mas_vision_replay PRIVATE vision_replay)
