cmake_minimum_required(VERSION 3.22)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(mas_vision_record LANGUAGES CXX)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "This tool targets Linux")
    endif()
    get_filename_component(VISION_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
    set(CMAKE_CXX_STANDARD 23)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    set(CMAKE_CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
        message(FATAL_ERROR "Use GCC >= 13 and its C++23 standard library")
    endif()
    if(NOT CMAKE_BUILD_TYPE)
        set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
    endif()
    if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
        set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
    endif()
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    include("${VISION_SOURCE_DIR}/cmake/BaseDependencies.cmake")
    include("${VISION_SOURCE_DIR}/cmake/LocalDependencies.cmake")
    find_package(SDL2 REQUIRED CONFIG)
    add_library(buff_options INTERFACE)
    target_include_directories(buff_options INTERFACE "${VISION_SOURCE_DIR}")
    add_subdirectory("${VISION_SOURCE_DIR}/third_party" third_party)
    add_subdirectory("${VISION_SOURCE_DIR}/rm_utils/io" io)
    add_subdirectory("${VISION_SOURCE_DIR}/rm_utils/display" display)
    add_subdirectory("${VISION_SOURCE_DIR}/hardware" hardware)
endif()

add_library(vision_session STATIC session.cpp)
target_link_libraries(vision_session PUBLIC buff_io ${BUFF_YAML_TARGET})
add_library(vision_record STATIC record.cpp)
target_link_libraries(vision_record PUBLIC vision_session hikcamera serial buff_display)
add_executable(mas_vision_record main.cpp)
target_link_libraries(mas_vision_record PRIVATE vision_record)
