cmake_minimum_required(VERSION 3.22)
project(mas_vision LANGUAGES CXX)
set(VISION_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
include(cmake/ProjectOptions.cmake)
# 默认不编译回归测试；部分检查需要模型、素材或显示环境。
option(BUILD_TESTING "Build regression checks (requires test media)" OFF)

include(cmake/Dependencies.cmake)

# 每层只管理自己的子目录，不递归搜索所有 CMakeLists.txt。
add_subdirectory(third_party)
add_subdirectory(rm_utils)
add_subdirectory(hardware)
add_subdirectory(auto_buff)
add_subdirectory(auto_aim)
add_subdirectory(threads)
add_subdirectory(apps)
include(CTest)
option(BUILD_TOOLS "Build manual diagnostic and profiling tools" OFF)
option(VISION_GPU_PROFILE_ONLY "Build only OpenVINO GPU profiling and execution checks" OFF)
if(VISION_GPU_PROFILE_ONLY AND (NOT BUILD_TESTING OR NOT YOLO_RUNTIME STREQUAL "OPENVINO"))
    message(FATAL_ERROR "VISION_GPU_PROFILE_ONLY requires BUILD_TESTING=ON and YOLO_RUNTIME=OPENVINO")
endif()
if(BUILD_TOOLS OR VISION_GPU_PROFILE_ONLY)
    add_subdirectory(tools)
endif()
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
