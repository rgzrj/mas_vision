cmake_minimum_required(VERSION 3.22)
project(mas_vision LANGUAGES CXX)
set(VISION_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "mas_vision now targets Linux only. Build on Ubuntu with GCC >= 13; see README_LINUX.md.")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
    set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# 分目录后仍把程序放在 build/，保持现有启动路径。
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND CMAKE_CXX_COMPILER_VERSION VERSION_LESS 13)
    message(FATAL_ERROR "RMCS needs GCC >= 13 and its C++23 standard library")
endif()
option(BUILD_TESTING "Build regression checks (requires test media)" OFF)

include(cmake/Dependencies.cmake)

# 自有模块显式链接；不修改第三方依赖的全局编译选项。
add_library(buff_options INTERFACE)
target_include_directories(buff_options INTERFACE "${PROJECT_SOURCE_DIR}")

# 每层只管理自己的子目录，不递归搜索所有 CMakeLists.txt。
add_subdirectory(third_party)
add_subdirectory(rm_utils)
add_subdirectory(hardware)
add_subdirectory(auto_buff)
add_subdirectory(auto_aim)
add_subdirectory(threads)
add_subdirectory(apps)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
